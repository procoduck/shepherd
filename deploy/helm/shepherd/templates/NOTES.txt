{{- /*
  The external URLs, one per hostname, each with the scheme that exposure
  really serves (#425 -- this printed https:// for every Ingress and
  HTTPRoute, TLS or not):

    Ingress    https when an ingress.tls entry covers the host (an exact name,
               a "*.parent" wildcard one label above it, or an entry with no
               hosts, which Kubernetes applies to every host); http otherwise.
    HTTPRoute  the Gateway listener decides, and the chart cannot read it
               without a lookup that a namespace-scoped deployer may not be
               allowed. route.scheme when set; otherwise http when the first
               parentRef names port 80, https when it names 443, and https
               assumed (with a hint saying so) when it names neither.

  The spoke URL collectors are told to use is the first https one that is not
  a wildcard (a wildcard is a pattern, not an address; collectors send their
  token over it, so TLS is preferred), else the first non-wildcard one; with
  none it is the in-cluster URL. Ranging over the hosts into one string once
  printed "https://a.example.comhttps://b.example.com".
*/ -}}
{{- $externalURLs := list -}}
{{- $externalURL := "" -}}
{{- $schemeAssumed := false -}}
{{- if and .Values.route.enabled .Values.route.hostnames -}}
{{- $scheme := .Values.route.scheme | default "" -}}
{{- if not $scheme -}}
{{- $scheme = "https" -}}
{{- $schemeAssumed = true -}}
{{- with .Values.route.parentRefs -}}
{{- $refPort := toString ((first .).port | default "") -}}
{{- if eq $refPort "80" -}}
{{- $scheme = "http" -}}
{{- $schemeAssumed = false -}}
{{- else if eq $refPort "443" -}}
{{- $schemeAssumed = false -}}
{{- end -}}
{{- end -}}
{{- end -}}
{{- range .Values.route.hostnames -}}
{{- $externalURLs = append $externalURLs (printf "%s://%s" $scheme .) -}}
{{- end -}}
{{- else if and .Values.ingress.enabled .Values.ingress.hosts -}}
{{- range .Values.ingress.hosts -}}
{{- $host := .host -}}
{{- $scheme := "http" -}}
{{- range $.Values.ingress.tls -}}
{{- if not .hosts -}}
{{- $scheme = "https" -}}
{{- end -}}
{{- range .hosts -}}
{{- $suffix := trimPrefix "*" . -}}
{{- if eq . $host -}}
{{- $scheme = "https" -}}
{{- else if and (hasPrefix "*." .) (hasSuffix $suffix $host) (not (contains "." (trimSuffix $suffix $host))) -}}
{{- $scheme = "https" -}}
{{- end -}}
{{- end -}}
{{- end -}}
{{- $externalURLs = append $externalURLs (printf "%s://%s" $scheme $host) -}}
{{- end -}}
{{- end -}}
{{- range $externalURLs -}}
{{- if and (not $externalURL) (hasPrefix "https://" .) (not (contains "://*" .)) -}}
{{- $externalURL = . -}}
{{- end -}}
{{- end -}}
{{- range $externalURLs -}}
{{- if and (not $externalURL) (not (contains "://*" .)) -}}
{{- $externalURL = . -}}
{{- end -}}
{{- end -}}

*/ -}}
